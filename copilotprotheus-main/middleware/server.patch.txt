// PASSO A — Adicionar require no topo do arquivo:
const { generateToken, verifyToken } = require('./auth/jwtMiddleware');

// PASSO B — Adicionar rota publica de geracao de token (antes das rotas protegidas):
app.post('/auth/token', (req, res) => {
  const { user, session_id, environment } = req.body;
  if (!user || !session_id)
    return res.status(400).json({ error: 'user e session_id sao obrigatorios.' });
  const token = generateToken({ user, session_id, environment: environment || 'validacao' });
  res.json({ token });
});
